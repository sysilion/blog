{{- /* 이 용어 문서를 [[..]] 로 가리키는 페이지 목록. MediaWiki 의 '여기를 가리키는 문서', Obsidian 의 Backlinks. 최신순. */ -}} {{- $target := . -}} {{- $names := partial "wiki/names.html" $target -}} {{- $found := slice -}} {{- range $p := site.RegularPages -}} {{- if ne $p.RelPermalink $target.RelPermalink -}} {{- /* targets 는 페이지별로 한 번만 계산한다 (용어 문서 수 × 전체 페이지 수 만큼 원문 정규식 스캔 방지). */ -}} {{- if intersect $names (partialCached "wiki/targets.html" $p $p.Path) -}} {{- $found = $found | append $p -}} {{- end -}} {{- end -}} {{- end -}} {{- return (sort $found "Date" "desc") -}}
