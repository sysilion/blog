{{- /* RawContent 에서 코드 펜스와 인라인 코드를 걷어낸 텍스트. 역링크·필요문서 스캔은 렌더된 HTML 이 아니라 원문을 보므로, 여기서 코드를 빼지 않으면 코드 예제 안의 [[..]] 를 용어 링크로 오인한다. */ -}} {{- $t := .RawContent -}} {{- $t = replaceRE "(?sm)^ {0,3}```.*?^ {0,3}```" "" $t -}} {{- $t = replaceRE "(?sm)^ {0,3}~~~.*?^ {0,3}~~~" "" $t -}} {{- $t = replaceRE "`[^`\n]*`" "" $t -}} {{- /* 표 셀 안에서는 [[slug\|표시명]] 으로 파이프를 이스케이프한다. goldmark 가 벗겨주는 건 렌더 결과뿐이라 원문 스캔에서는 직접 벗긴다. */ -}} {{- $t = replace $t "\\|" "|" -}} {{- return $t -}}
